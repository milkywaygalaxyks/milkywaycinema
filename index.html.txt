  <script>
    function convertDriveLink(link) {
      const match = link.match(/\/d\/(.+?)\//);
      return match ? `https://drive.google.com/file/d/${match[1]}/preview` : link;
    }

    const movies = [

      { id: 3, title: "SIR (2024)", poster: "Posters/Poster3.jpg", video: convertDriveLink("https://drive.google.com/file/d/1XHZh-oNHQriNF0H0M9jfA4XoVZuSEpIE/view?usp=sharing") },
      { id: 4, title: "3BHK (2025)", poster: "Posters/Poster4.jpg", video: convertDriveLink("https://drive.google.com/file/d/16vrFBJ3UbtT2e4ht-lf2QBpoVGqbF8zX/view?usp=sharing") },
      { id: 5, title: "Hridayam (2022)", poster: "Posters/Poster5.jpg", video: "https://youtu.be/fJftRAwst3w" },
      { id: 6, title: "Kingston (2025)", poster: "Posters/Poster6.jpg", video: convertDriveLink("https://drive.google.com/file/d/13yKXv-aVe_UbVpUb1uvOmpFav69ANKob/view?usp=sharing") },
      { id: 7, title: "Idly Kadai (2025)", poster: "Posters/Poster7.jpg", video: convertDriveLink("https://drive.google.com/file/d/1lnpKx3YsEdQV8MfKBjvrrGCKln7U6rFV/view?usp=sharing") }
    ];

    localStorage.setItem("movies", JSON.stringify(movies));

    const container = document.getElementById("movieContainer");
    const searchInput = document.getElementById("searchInput");

    function displayMovies(filter = "") {
      container.innerHTML = "";
      movies
        .filter(movie => movie.title.toLowerCase().includes(filter.toLowerCase()))
        .forEach(movie => {
          const card = document.createElement("div");
          card.className = "movie-card";
          card.tabIndex = 0; 

          card.innerHTML = `
            <img src="${movie.poster}" alt="${movie.title}">
            <h3>${movie.title}</h3>
          `;
          
          card.onclick = () => {
            window.location.href = `movie.html?id=${movie.id}`;
          };
          
          card.addEventListener("keydown", function(event) {
            if (event.key === "Enter") {
              window.location.href = `movie.html?id=${movie.id}`;
            }
          });

          container.appendChild(card);
        });
    }

    displayMovies();

    searchInput.addEventListener("input", (e) => {
      displayMovies(e.target.value);
    });

    // --- 🖱️ VIRTUAL TV MOUSE CURSOR ---
    const cursor = document.createElement("div");
    cursor.id = "tv-cursor";
    document.body.appendChild(cursor);

    let cx = window.innerWidth / 2;
    let cy = window.innerHeight / 2;
    let cursorSpeed = 15; 
    let keysPressed = {};

    function updateCursor() {
      cursor.style.left = cx + "px";
      cursor.style.top = cy + "px";
    }
    updateCursor();

    window.addEventListener("keydown", (e) => {
      keysPressed[e.key] = true;

      if (e.key === "Enter") {
        cursor.style.display = 'none';
        const elementBelow = document.elementFromPoint(cx, cy);
        cursor.style.display = 'block';

        if (elementBelow) {
          elementBelow.click();
          elementBelow.focus(); 
        }
      }
    });

    window.addEventListener("keyup", (e) => {
      keysPressed[e.key] = false;
    });

    function moveCursorLoop() {
      let moved = false;

      if (keysPressed["ArrowUp"]) { cy -= cursorSpeed; moved = true; }
      if (keysPressed["ArrowDown"]) { cy += cursorSpeed; moved = true; }
      if (keysPressed["ArrowLeft"]) { cx -= cursorSpeed; moved = true; }
      if (keysPressed["ArrowRight"]) { cx += cursorSpeed; moved = true; }

      if (moved) {
        cx = Math.max(0, Math.min(window.innerWidth - 25, cx));
        cy = Math.max(0, Math.min(window.innerHeight - 25, cy));
        
        if (cy > window.innerHeight - 50) window.scrollBy(0, cursorSpeed);
        if (cy < 50) window.scrollBy(0, -cursorSpeed);

        updateCursor();
      }
      requestAnimationFrame(moveCursorLoop);
    }
    moveCursorLoop();
